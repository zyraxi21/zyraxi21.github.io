## PowerShell

```powershell
# 汇总日志目录中占用空间最大的若干个文件
[CmdletBinding()]
param(
    [Parameter(Mandatory = $true)]
    [string]$Path,

    [int]$Top = 5
)

$ErrorActionPreference = "Stop"
$files = Get-ChildItem -Path $Path -Filter *.log -Recurse |
         Where-Object { $_.Length -gt 1KB } |
         Sort-Object Length -Descending |
         Select-Object -First $Top

foreach ($f in $files) {
    $sizeMB = [math]::Round($f.Length / 1MB, 2)
    Write-Host "文件 $($f.Name) 大小 ${sizeMB}MB" -ForegroundColor Green
}

if ($files.Count -eq 0) {
    throw "在 $Path 下没有找到符合条件的日志文件"
}
```

## Python

```python
#!/usr/bin/env python3
"""按大小筛选日志文件并汇总统计。"""
import os
import sys
from dataclasses import dataclass

THRESHOLD = 1024 * 1024  # 1 MiB


@dataclass
class Entry:
    name: str
    size: int

    @property
    def size_mb(self) -> float:
        return round(self.size / THRESHOLD, 2)


def scan(root: str):
    for dirpath, _, filenames in os.walk(root):
        for fn in filenames:
            if fn.endswith(".log"):
                yield Entry(fn, os.path.getsize(os.path.join(dirpath, fn)))


def main() -> int:
    entries = sorted(scan(sys.argv[1] if len(sys.argv) > 1 else "."),
                     key=lambda e: e.size, reverse=True)
    if not entries:
        print("没有找到 .log 文件", file=sys.stderr)
        return 1

    for i, e in enumerate(entries[:5], start=1):
        print(f"{i:>2}. {e.name:<24} {e.size_mb:>8.2f} MB")

    total = sum(e.size for e in entries)
    print(f"共 {len(entries)} 个文件，合计 {total / THRESHOLD:.2f} MB")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

## MATLAB

```matlab
% 计算雷达方程的最大探测距离
function R_max = radarRange(Pt, G, lambda, sigma, Pmin, L)
% RADARRANGE 输入发射功率、增益、波长、RCS、最小可检测功率与损耗
    if nargin < 6
        L = 1.0;              % 默认无损耗
    end

    const = 4 * pi;
    R_max = ((Pt * G^2 * lambda^2 * sigma) / ...
             (const^2 * Pmin * L))^(1/4);

    fprintf('最大探测距离 = %.2f km\n', R_max / 1000);
end

% 批量扫描并绘图
Pt = 1e6;          % 峰值功率 1 MW
G  = 10^(35/10);   % 增益 35 dB
lambda = 3e8 / 3e9;
sigma  = [1 5 10];

for k = 1:numel(sigma)
    R(k) = radarRange(Pt, G, lambda, sigma(k), 1e-13);
    assert(R(k) > 0, 'R 必须为正');
end

semilogx(sigma, R / 1000, '-o', 'LineWidth', 1.5);
xlabel('RCS / m^2'); ylabel('R_{max} / km');
grid on; title('探测距离');
```

## C

```c
#include <stdio.h>
#include <stdlib.h>

#define MAX_NAME 64
#define SQUARE(x) ((x) * (x))

typedef struct {
    char name[MAX_NAME];
    unsigned int id;
    double score;
} Student;

static int compare_by_score(const void *a, const void *b)
{
    const Student *sa = (const Student *)a;
    const Student *sb = (const Student *)b;
    if (sa->score < sb->score) return  1;
    if (sa->score > sb->score) return -1;
    return 0;
}

int main(void)
{
    Student list[] = {
        { "Alice", 1001, 92.5 },
        { "Bob",   1002, 88.0 },
    };
    size_t n = sizeof(list) / sizeof(list[0]);

    qsort(list, n, sizeof(Student), compare_by_score);

    for (size_t i = 0; i < n; ++i) {
        printf("%-8s %5u %6.2f %ld\n",
               list[i].name, list[i].id, list[i].score, SQUARE(i));
    }
    return 0;
}
```

## Kotlin

```kotlin
package com.example.radar

import kotlin.math.sqrt

/** 单个目标点。 */
data class Target(val name: String, val rcs: Double, val range: Double)

const val MAX_RANGE_KM = 300.0

class Radar(private val gain: Double, private val wavelength: Double) {

    private val targets = mutableListOf<Target>()

    fun add(target: Target) = apply { targets += target }

    fun detectable(power: Double): List<Target> = targets.filter { t ->
        val r = sqrt(sqrt(power * gain * wavelength * wavelength * t.rcs / 1e-13))
        r <= MAX_RANGE_KM
    }

    companion object {
        const val VERSION: String = "1.0.0"
    }
}

fun main() {
    val radar = Radar(gain = 10.0, wavelength = 0.1)
        .add(Target("Alpha", 1.0, 120.0))
        .add(Target("Bravo", 2.5, 260.0))

    val hits = radar.detectable(1e6)
    println("命中 ${hits.size} 个目标")

    val total = (1..5).map { it * 2 }.sum()
    println("总计 $total，版本 ${Radar.VERSION}")
}
```

# 公式测试
由 $\nabla \boldsymbol{\cdot B}=0$ 知存在矢量函数 $\boldsymbol{A}$，使得$$\boldsymbol{B}=\nabla \times \boldsymbol{A}$$称 $\boldsymbol{A}$ 为动态矢量磁位（磁矢位）。

将上式代入 $\nabla \times \boldsymbol{E}=- \frac{\partial \boldsymbol{B}}{\partial t}$，得 $\nabla \times\left( \boldsymbol{E}+ \frac{\partial \boldsymbol{A}}{\partial t} \right)=0$，则存在标量函数 $\phi$，使得$$\boldsymbol{E}+ \frac{\partial \boldsymbol{A}}{\partial t}=-\nabla \phi$$称 $\phi$ 为标量电位（电标位）。

为确定矢量函数 $\boldsymbol{A}$，令$$\boxed{\nabla \boldsymbol{\cdot A}+\mu\varepsilon  \frac{\partial \phi}{\partial t}=0}$$上式称为**洛伦兹规范（洛伦兹条件）**，则得到达朗贝尔方程（达朗伯方程）：$$\nabla^{2}\phi-\mu\varepsilon  \frac{\partial^{2}\phi}{\partial t^{2}}=- \frac{\rho}{\varepsilon}$$ $$\nabla^{2}\boldsymbol{A}-\mu\varepsilon  \frac{\partial^{2}\boldsymbol{A}}{\partial t^{2}}=-\mu \boldsymbol{J}$$
> 洛伦兹规范在静态场的条件下退化为库仑规范。

对于时谐场，有非齐次亥姆霍兹方程$$\begin{cases}\nabla^{2}\phi+k^{2}\phi=- \frac{\rho}{\varepsilon}\\\nabla^{2}\boldsymbol{A}+k^{2}\boldsymbol{A}=-\mu \boldsymbol{J}\end{cases}$$洛伦兹规范为$$\boxed{\nabla \boldsymbol{\cdot A}=-\mathrm{j}\omega \mu \varepsilon \phi}$$
